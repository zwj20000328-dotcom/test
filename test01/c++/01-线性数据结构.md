## 数组 (Array)

### 基本定义
```cpp
// 静态数组
int arr[10];                    // 未初始化，元素值不确定
int arr[5] = {1, 2, 3, 4, 5};  // 完全初始化
int arr[5] = {1, 2, 3};        // 部分初始化，剩余为0
int arr[] = {1, 2, 3, 4};      // 自动推导大小为4

// 多维数组
int matrix[3][4];              // 3x4 二维数组
int cube[2][3][4];             // 2x3x4 三维数组
```

### 常用操作
```cpp
// 遍历
for (int i = 0; i < n; i++) {
    cout << arr[i] << " ";
}

// 范围 for 循环（C++11）
for (int x : arr) {
    cout << x << " ";
}

// 求数组大小
int size = sizeof(arr) / sizeof(arr[0]);

// 填充数组
memset(arr, 0, sizeof(arr));   // 填充为0（需 #include <cstring>）
fill(arr, arr + n, value);     // 填充为指定值（需 #include <algorithm>）

// 排序
sort(arr, arr + n);            // 升序
sort(arr, arr + n, greater<int>()); // 降序

// 查找
int* ptr = find(arr, arr + n, target);
if (ptr != arr + n) {
    int index = ptr - arr;     // 找到，返回索引
}

// 反转
reverse(arr, arr + n);
```

### 注意事项
- 大小在编译时确定，不能动态改变
- 数组名是指向首元素的指针
- 传递给函数时会退化为指针，丢失大小信息
- 越界访问不会报错，可能导致未定义行为

## 向量 (Vector)

### 头文件
```cpp
#include <vector>
```

### 创建与初始化
```cpp
vector<int> v;                          // 空向量
vector<int> v(5);                       // 5个元素，默认值为0
vector<int> v(5, 10);                   // 5个元素，值都为10
vector<int> v = {1, 2, 3, 4, 5};       // 列表初始化
vector<int> v{1, 2, 3, 4, 5};          // 列表初始化
vector<int> v2(v1);                     // 拷贝构造
vector<int> v2 = v1;                    // 拷贝构造
vector<int> v(arr, arr + n);           // 从数组初始化
vector<vector<int>> matrix(3, vector<int>(4)); // 二维向量 3x4
```

### 容量相关操作
```cpp
v.size();              // 返回元素个数          O(1)
v.empty();             // 判断是否为空          O(1)
v.capacity();          // 返回当前容量          O(1)
v.max_size();          // 返回最大可容纳元素数  O(1)
v.reserve(n);          // 预分配至少n个元素空间 O(n)
v.resize(n);           // 改变大小为n           O(n)
v.resize(n, val);      // 改变大小为n，新增元素为val
v.shrink_to_fit();     // 释放多余容量          O(n)
```

### 访问元素
```cpp
v[i];                  // 访问第i个元素（不检查越界）  O(1)
v.at(i);              // 访问第i个元素（检查越界）    O(1)
v.front();            // 访问第一个元素               O(1)
v.back();             // 访问最后一个元素             O(1)
v.data();             // 返回指向底层数组的指针       O(1)
```

### 修改操作
```cpp
v.push_back(val);     // 尾部添加元素          均摊 O(1)
v.emplace_back(val);  // 尾部就地构造元素      均摊 O(1)（更高效）
v.pop_back();         // 删除尾部元素          O(1)
v.insert(pos, val);   // 在pos位置插入元素     O(n)
v.insert(pos, n, val); // 在pos位置插入n个val  O(n)
v.insert(pos, first, last); // 插入区间        O(n)
v.emplace(pos, val);  // 在pos位置就地构造     O(n)
v.erase(pos);         // 删除pos位置元素       O(n)
v.erase(first, last); // 删除区间元素          O(n)
v.clear();            // 清空所有元素          O(n)
v.assign(n, val);     // 重新赋值为n个val      O(n)
v.assign(first, last); // 重新赋值为区间       O(n)
v.swap(v2);           // 交换两个向量          O(1)
swap(v1, v2);         // 交换两个向量          O(1)
```

### 迭代器操作
```cpp
v.begin();            // 返回首元素迭代器
v.end();              // 返回尾后迭代器
v.rbegin();           // 返回逆向首元素迭代器
v.rend();             // 返回逆向尾后迭代器
v.cbegin();           // 返回常量首元素迭代器
v.cend();             // 返回常量尾后迭代器

// 遍历方式
for (auto it = v.begin(); it != v.end(); ++it) {
    cout << *it << " ";
}

for (auto it = v.rbegin(); it != v.rend(); ++it) {
    cout << *it << " ";
}

for (int i = 0; i < v.size(); i++) {
    cout << v[i] << " ";
}

for (const auto& x : v) {
    cout << x << " ";
}
```

### 算法配合
```cpp
sort(v.begin(), v.end());                    // 升序排序
sort(v.begin(), v.end(), greater<int>());    // 降序排序
reverse(v.begin(), v.end());                 // 反转
auto it = find(v.begin(), v.end(), target);  // 查找
int cnt = count(v.begin(), v.end(), val);    // 计数
v.erase(unique(v.begin(), v.end()), v.end()); // 去重（需先排序）
int sum = accumulate(v.begin(), v.end(), 0); // 求和
auto [min_it, max_it] = minmax_element(v.begin(), v.end()); // 最值
```

### 性能特点
- 随机访问：O(1)
- 尾部插入/删除：均摊 O(1)
- 中间插入/删除：O(n)
- 查找：O(n)（无序），O(log n)（有序，用二分）
- 内存连续，缓存友好

 ## 链表 (Linked List)

### 手动实现单向链表
```cpp
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

// 创建链表
ListNode* head = new ListNode(1);
head->next = new ListNode(2);
head->next->next = new ListNode(3);

// 头部插入
ListNode* newNode = new ListNode(val);
newNode->next = head;
head = newNode;

// 尾部插入
ListNode* curr = head;
while (curr->next) curr = curr->next;
curr->next = new ListNode(val);

// 指定位置后插入
ListNode* newNode = new ListNode(val);
newNode->next = curr->next;
curr->next = newNode;

// 删除头部
ListNode* temp = head;
head = head->next;
delete temp;

// 删除指定节点后一个节点
ListNode* temp = curr->next;
curr->next = curr->next->next;
delete temp;

// 遍历
ListNode* curr = head;
while (curr) {
    cout << curr->val << " ";
    curr = curr->next;
}

// 释放链表
while (head) {
    ListNode* temp = head;
    head = head->next;
    delete temp;
}
```

### STL list（双向链表）
```cpp
#include <list>

list<int> lst;                    // 空链表
list<int> lst(5);                 // 5个元素，默认值为0
list<int> lst(5, 10);             // 5个元素，值为10
list<int> lst = {1, 2, 3, 4, 5}; // 列表初始化
```

### list 常用操作
```cpp
// 容量
lst.size();                       // 元素个数
lst.empty();                      // 是否为空

// 访问（不支持随机访问）
lst.front();                      // 首元素
lst.back();                       // 尾元素

// 修改
lst.push_back(val);               // 尾部添加
lst.push_front(val);              // 头部添加
lst.emplace_back(val);            // 尾部就地构造
lst.emplace_front(val);           // 头部就地构造
lst.pop_back();                   // 删除尾部
lst.pop_front();                  // 删除头部
lst.insert(pos, val);             // 在pos位置插入
lst.insert(pos, n, val);          // 插入n个val
lst.erase(pos);                   // 删除pos位置元素
lst.erase(first, last);           // 删除区间
lst.clear();                      // 清空

// 特有操作
lst.remove(val);                  // 删除所有等于val的元素  O(n)
lst.remove_if(pred);              // 删除满足条件的元素    O(n)
lst.unique();                     // 去重（需先排序）      O(n)
lst.sort();                       // 排序                  O(n log n)
lst.reverse();                    // 反转                  O(n)
lst.merge(lst2);                  // 合并两个有序链表      O(n)
lst.splice(pos, lst2);            // 将lst2插入到pos位置   O(1)
lst.splice(pos, lst2, pos2);      // 移动lst2中pos2元素    O(1)
lst.swap(lst2);                   // 交换                  O(1)
```

### forward_list（单向链表，C++11）
```cpp
#include <forward_list>

forward_list<int> flst = {1, 2, 3};

flst.push_front(val);             // 头部插入
flst.emplace_front(val);          // 头部就地构造
flst.pop_front();                 // 删除头部
flst.front();                     // 访问头部
flst.before_begin();              // 返回首前迭代器
flst.insert_after(pos, val);      // 在pos后插入
flst.erase_after(pos);            // 删除pos后元素
flst.remove(val);                 // 删除等于val的元素
flst.reverse();                   // 反转
flst.sort();                      // 排序
flst.unique();                    // 去重
```

### 性能对比
| 操作 | vector | list | forward_list |
|------|--------|------|--------------|
| 随机访问 | O(1) | O(n) | O(n) |
| 头部插入/删除 | O(n) | O(1) | O(1) |
| 尾部插入/删除 | 均摊O(1) | O(1) | O(n) |
| 中间插入/删除 | O(n) | O(1)* | O(1)* |
| 内存占用 | 较少 | 较多 | 中等 |

*需已知位置


### 底层容器选择
```cpp
stack<int, vector<int>> st1;      // 底层用vector
stack<int, deque<int>> st2;       // 底层用deque（默认）
stack<int, list<int>> st3;        // 底层用list
```

### 手动实现栈
```cpp
class Stack {
private:
    int* data;
    int top_idx;
    int capacity;
public:
    Stack(int cap = 100) : capacity(cap), top_idx(-1) {
        data = new int[capacity];
    }
    ~Stack() { delete[] data; }
    
    void push(int val) {
        if (top_idx < capacity - 1) data[++top_idx] = val;
    }
    void pop() {
        if (top_idx >= 0) top_idx--;
    }
    int top() {
        if (top_idx >= 0) return data[top_idx];
        return -1; // 或抛异常
    }
    bool empty() { return top_idx == -1; }
    int size() { return top_idx + 1; }
};
```

### 应用场景
- 函数调用栈
- 表达式求值与括号匹配
- 深度优先搜索（DFS）
- 撤销操作
- 单调栈问题

### 手动实现循环队列
```cpp
class CircularQueue {
private:
    int* data;
    int front_idx, rear_idx;
    int capacity;
    int count;
public:
    CircularQueue(int cap) : capacity(cap + 1), front_idx(0), rear_idx(0), count(0) {
        data = new int[capacity];
    }
    ~CircularQueue() { delete[] data; }
    
    void push(int val) {
        if (count < capacity - 1) {
            data[rear_idx] = val;
            rear_idx = (rear_idx + 1) % capacity;
            count++;
        }
    }
    void pop() {
        if (count > 0) {
            front_idx = (front_idx + 1) % capacity;
            count--;
        }
    }
    int front() { return count > 0 ? data[front_idx] : -1; }
    bool empty() { return count == 0; }
    int size() { return count; }
};
```

### 应用场景
- 广度优先搜索（BFS）
- 任务调度
- 缓冲区
- 层序遍历

## 双端队列 (Deque)

### STL deque
```cpp
#include <deque>

deque<int> dq;                    // 创建空双端队列

// 修改操作
dq.push_back(val);                // 尾部添加      均摊 O(1)
dq.push_front(val);               // 头部添加      均摊 O(1)
dq.emplace_back(val);             // 尾部就地构造  均摊 O(1)
dq.emplace_front(val);            // 头部就地构造  均摊 O(1)
dq.pop_back();                    // 删除尾部      O(1)
dq.pop_front();                   // 删除头部      O(1)
dq.insert(pos, val);              // 在pos位置插入 O(n)
dq.erase(pos);                    // 删除pos元素   O(n)
dq.clear();                       // 清空          O(n)

// 访问操作
dq[i];                            // 随机访问      O(1)
dq.at(i);                         // 带检查访问    O(1)
dq.front();                       // 首元素        O(1)
dq.back();                        // 尾元素        O(1)

// 容量
dq.size();                        // 元素个数      O(1)
dq.empty();                       // 是否为空      O(1)
dq.resize(n);                     // 改变大小      O(n)
```

### 特点
- 支持随机访问（类似vector）
- 头部和尾部都能高效插入/删除
- 内存不连续（分段存储）
- 迭代器稳定性较好（插入/删除后大部分迭代器仍有效）

### 应用场景
- 滑动窗口问题
- 需要两端操作的场景
- 作为stack和queue的底层容器

## 字符串 (String)

### 头文件
```cpp
#include <string>
```

### 创建与初始化
```cpp
string s;                         // 空字符串
string s("hello");                // 从C字符串
string s = "hello";               // 从C字符串
string s(5, 'a');                 // 5个'a'，结果为"aaaaa"
string s(other);                  // 拷贝构造
string s(other, 2, 3);            // 从other的索引2开始取3个字符
string s(arr, 5);                 // 从字符数组前5个字符
```

### 常用操作
```cpp
// 容量
s.size();                         // 字符个数      O(1)
s.length();                       // 同size()      O(1)
s.empty();                        // 是否为空      O(1)
s.capacity();                     // 当前容量      O(1)
s.reserve(n);                     // 预分配空间    O(n)
s.resize(n);                      // 改变大小      O(n)
s.resize(n, ch);                  // 用ch填充新增
s.shrink_to_fit();                // 释放多余容量
s.max_size();                     // 最大可容纳数

// 访问
s[i];                             // 访问第i个字符  O(1)
s.at(i);                          // 带检查访问     O(1)
s.front();                        // 首字符（C++11）O(1)
s.back();                         // 尾字符（C++11）O(1)
s.c_str();                        // 转C字符串      O(1)
s.data();                         // 返回字符指针   O(1)

// 修改
s.push_back(ch);                  // 尾部添加字符   均摊 O(1)
s.pop_back();                     // 删除尾部字符   O(1)
s += str;                         // 追加字符串     均摊 O(n)
s += ch;                          // 追加字符       均摊 O(1)
s.append(str);                    // 追加字符串     均摊 O(n)
s.append(str, pos, len);          // 追加子串
s.append(n, ch);                  // 追加n个ch
s.insert(pos, str);               // 在pos插入字符串 O(n)
s.insert(pos, n, ch);             // 插入n个ch      O(n)
s.erase(pos, len);                // 删除len个字符  O(n)
s.replace(pos, len, str);         // 替换           O(n)
s.clear();                        // 清空           O(n)
s.swap(s2);                       // 交换           O(1)
```
