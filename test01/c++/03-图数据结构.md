## 图的表示

### 邻接矩阵
```cpp
// 无权图
int adj[MAXN][MAXN]; // adj[u][v] = 1 表示有边

// 带权图
int adj[MAXN][MAXN]; // adj[u][v] = w 表示边权为w，无边时为INF
memset(adj, 0x3f, sizeof(adj)); // 初始化为INF

// 添加边（无向图）
void addEdge(int u, int v) {
    adj[u][v] = 1;
    adj[v][u] = 1;
}

// 添加边（有向图）
void addEdge(int u, int v) {
    adj[u][v] = 1;
}

// 空间复杂度：O(V²)
// 适用场景：稠密图、需要快速判断两点间是否有边
```

## 图的遍历

### BFS（广度优先搜索）
```cpp
// 基础BFS
vector<int> bfs(int n, vector<int> adj[], int start) {
    vector<int> order;
    vector<bool> visited(n + 1, false);
    queue<int> q;
    
    visited[start] = true;
    q.push(start);
    
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        order.push_back(u);
        
        for (int v : adj[u]) {
            if (!visited[v]) {
                visited[v] = true;
                q.push(v);
            }
        }
    }
    return order;
}

// BFS求最短路（无权图）
vector<int> bfsShortestPath(int n, vector<int> adj[], int start) {
    vector<int> dist(n + 1, -1);
    queue<int> q;
    
    dist[start] = 0;
    q.push(start);
    
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        
        for (int v : adj[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
    return dist;
}

// BFS路径还原
vector<int> bfsPath(int n, vector<int> adj[], int start, int end) {
    vector<int> parent(n + 1, -1);
    vector<bool> visited(n + 1, false);
    queue<int> q;
    
    visited[start] = true;
    q.push(start);
    
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (u == end) break;
        
        for (int v : adj[u]) {
            if (!visited[v]) {
                visited[v] = true;
                parent[v] = u;
                q.push(v);
            }
        }
    }
    
    vector<int> path;
    for (int v = end; v != -1; v = parent[v]) {
        path.push_back(v);
    }
    reverse(path.begin(), path.end());
    return path;
}
```

### DFS（深度优先搜索）
```cpp
// 递归DFS
void dfs(int u, vector<int> adj[], vector<bool>& visited, vector<int>& order) {
    visited[u] = true;
    order.push_back(u);
    for (int v : adj[u]) {
        if (!visited[v]) {
            dfs(v, adj, visited, order);
        }
    }
}

// 迭代DFS
vector<int> dfsIterative(int n, vector<int> adj[], int start) {
    vector<int> order;
    vector<bool> visited(n + 1, false);
    stack<int> st;
    
    st.push(start);
    
    while (!st.empty()) {
        int u = st.top();
        st.pop();
        
        if (visited[u]) continue;
        visited[u] = true;
        order.push_back(u);
        
        for (int i = adj[u].size() - 1; i >= 0; i--) {
            int v = adj[u][i];
            if (!visited[v]) {
                st.push(v);
            }
        }
    }
    return order;
}

// 连通分量计数
int countComponents(int n, vector<int> adj[]) {
    vector<bool> visited(n + 1, false);
    int count = 0;
    for (int i = 1; i <= n; i++) {
        if (!visited[i]) {
            dfs(i, adj, visited);
            count++;
        }
    }
    return count;
}

// 检测环（无向图）
bool hasCycleUndirected(int n, vector<int> adj[]) {
    vector<bool> visited(n + 1, false);
    for (int i = 1; i <= n; i++) {
        if (!visited[i]) {
            if (dfsCycle(i, adj, visited, -1)) return true;
        }
    }
    return false;
}
bool dfsCycle(int u, vector<int> adj[], vector<bool>& visited, int parent) {
    visited[u] = true;
    for (int v : adj[u]) {
        if (!visited[v]) {
            if (dfsCycle(v, adj, visited, u)) return true;
        } else if (v != parent) {
            return true;
        }
    }
    return false;
}

// 检测环（有向图）
bool hasCycleDirected(int n, vector<int> adj[]) {
    vector<int> color(n + 1, 0); // 0:白 1:灰 2:黑
    for (int i = 1; i <= n; i++) {
        if (color[i] == 0) {
            if (dfsCycleDirected(i, adj, color)) return true;
        }
    }
    return false;
}
bool dfsCycleDirected(int u, vector<int> adj[], vector<int>& color) {
    color[u] = 1;
    for (int v : adj[u]) {
        if (color[v] == 1) return true;
        if (color[v] == 0 && dfsCycleDirected(v, adj, color)) return true;
    }
    color[u] = 2;
    return false;
}
```

## 最短路径算法

### Dijkstra（单源最短路，非负权）
```cpp
// 时间复杂度：O((V + E) log V)
vector<long long> dijkstra(int n, vector<pair<int, int>> adj[], int start) {
    const long long INF = 1e18;
    vector<long long> dist(n + 1, INF);
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<>> pq;
    
    dist[start] = 0;
    pq.push({0, start});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        
        if (d > dist[u]) continue;
        
        for (auto& [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}

// 带路径还原
pair<vector<long long>, vector<int>> dijkstraWithPath(int n, vector<pair<int, int>> adj[], int start) {
    const long long INF = 1e18;
    vector<long long> dist(n + 1, INF);
    vector<int> parent(n + 1, -1);
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<>> pq;
    
    dist[start] = 0;
    pq.push({0, start});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d > dist[u]) continue;
        
        for (auto& [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                parent[v] = u;
                pq.push({dist[v], v});
            }
        }
    }
    return {dist, parent};
}
```

### Bellman-Ford（支持负权）
```cpp
// 时间复杂度：O(VE)
vector<long long> bellmanFord(int n, vector<Edge>& edges, int start) {
    const long long INF = 1e18;
    vector<long long> dist(n + 1, INF);
    dist[start] = 0;
    
    for (int i = 0; i < n - 1; i++) {
        for (auto& e : edges) {
            if (dist[e.from] != INF && dist[e.from] + e.weight < dist[e.to]) {
                dist[e.to] = dist[e.from] + e.weight;
            }
        }
    }
    
    // 检测负环
    for (auto& e : edges) {
        if (dist[e.from] != INF && dist[e.from] + e.weight < dist[e.to]) {
            return {}; // 存在负环
        }
    }
    return dist;
}
```

### SPFA（队列优化Bellman-Ford）
```cpp
// 平均时间复杂度：O(kE)，最坏O(VE)
vector<long long> spfa(int n, vector<pair<int, int>> adj[], int start) {
    const long long INF = 1e18;
    vector<long long> dist(n + 1, INF);
    vector<bool> inQueue(n + 1, false);
    vector<int> count(n + 1, 0);
    queue<int> q;
    
    dist[start] = 0;
    q.push(start);
    inQueue[start] = true;
    
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        inQueue[u] = false;
        
        for (auto& [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                if (!inQueue[v]) {
                    q.push(v);
                    inQueue[v] = true;
                    if (++count[v] >= n) return {}; // 负环
                }
            }
        }
    }
    return dist;
}
```

### Floyd-Warshall（全源最短路）
```cpp
// 时间复杂度：O(V³)
void floyd(int n, vector<vector<long long>>& dist) {
    for (int k = 1; k <= n; k++) {
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (dist[i][k] != INF && dist[k][j] != INF) {
                    dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
                }
            }
        }
    }
}

// 初始化

for (int i = 1; i <= n; i++) dist[i][i] = 0;
for (auto& e : edges) dist[e.from][e.to] = e.weight;
```

## 最小生成树

### Kruskal算法
```cpp
// 时间复杂度：O(E log E)
struct Edge {
    int from, to, weight;
    bool operator<(const Edge& other) const {
        return weight < other.weight;
    }
};

int kruskal(int n, vector<Edge>& edges) {
    sort(edges.begin(), edges.end());
    UnionFind uf(n + 1);
    int totalWeight = 0;
    int edgeCount = 0;
    
    for (auto& e : edges) {
        if (uf.unite(e.from, e.to)) {
            totalWeight += e.weight;
            edgeCount++;
            if (edgeCount == n - 1) break;
        }
    }
    return edgeCount == n - 1 ? totalWeight : -1; // -1表示不连通
}
```

### Prim算法
```cpp
// 时间复杂度：O((V + E) log V)
int prim(int n, vector<pair<int, int>> adj[]) {
    vector<bool> inMST(n + 1, false);
    vector<int> minEdge(n + 1, INF);
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> pq;
    
    minEdge[1] = 0;
    pq.push({0, 1});
    int totalWeight = 0;
    int count = 0;
    
    while (!pq.empty() && count < n) {
        auto [w, u] = pq.top();
        pq.pop();
        
        if (inMST[u]) continue;
        inMST[u] = true;
        totalWeight += w;
        count++;
        
        for (auto& [v, weight] : adj[u]) {
            if (!inMST[v] && weight < minEdge[v]) {
                minEdge[v] = weight;
                pq.push({weight, v});
            }
        }
    }
    return count == n ? totalWeight : -1;
}
```

## 拓扑排序

### BFS拓扑排序（Kahn算法）
```cpp
// 时间复杂度：O(V + E)
vector<int> topoSortBFS(int n, vector<int> adj[]) {
    vector<int> inDegree(n + 1, 0);
    for (int u = 1; u <= n; u++) {
        for (int v : adj[u]) {
            inDegree[v]++;
        }
    }
    
    queue<int> q;
    for (int i = 1; i <= n; i++) {
        if (inDegree[i] == 0) q.push(i);
    }
    
    vector<int> order;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        order.push_back(u);
        
        for (int v : adj[u]) {
            if (--inDegree[v] == 0) {
                q.push(v);
            }
        }
    }
    
    return order.size() == n ? order : vector<int>{}; // 空表示有环
}
```

### DFS拓扑排序
```cpp
vector<int> topoSortDFS(int n, vector<int> adj[]) {
    vector<int> color(n + 1, 0);
    vector<int> order;
    
    function<bool(int)> dfs = [&](int u) -> bool {
        color[u] = 1;
        for (int v : adj[u]) {
            if (color[v] == 1) return false;
            if (color[v] == 0 && !dfs(v)) return false;
        }
        color[u] = 2;
        order.push_back(u);
        return true;
    };
    
    for (int i = 1; i <= n; i++) {
        if (color[i] == 0 && !dfs(i)) return {};
    }
    reverse(order.begin(), order.end());
    return order;
}
```

## 强连通分量

### Tarjan算法
```cpp
// 时间复杂度：O(V + E)
struct Tarjan {
    int n;
    vector<vector<int>> adj;
    vector<int> dfn, low, sccId;
    vector<bool> inStack;
    stack<int> st;
    int timer, sccCount;
    vector<vector<int>> sccs;
    
    Tarjan(int n) : n(n), adj(n + 1), dfn(n + 1, 0), low(n + 1, 0),
                    sccId(n + 1, 0), inStack(n + 1, false),
                    timer(0), sccCount(0) {}
    
    void dfs(int u) {
        dfn[u] = low[u] = ++timer;
        st.push(u);
        inStack[u] = true;
        
        for (int v : adj[u]) {
            if (!dfn[v]) {
                dfs(v);
                low[u] = min(low[u], low[v]);
            } else if (inStack[v]) {
                low[u] = min(low[u], dfn[v]);
            }
        }
        
        if (low[u] == dfn[u]) {
            sccCount++;
            vector<int> scc;
            while (true) {
                int v = st.top();
                st.pop();
                inStack[v] = false;
                sccId[v] = sccCount;
                scc.push_back(v);
                if (v == u) break;
            }
            sccs.push_back(scc);
        }
    }
    
    void run() {
        for (int i = 1; i <= n; i++) {
            if (!dfn[i]) dfs(i);
        }
    }
};
```

## 网络流

### 最大流（Edmonds-Karp / BFS增广）
```cpp
struct MaxFlow {
    int n;
    struct Edge {
        int to, rev;
        long long cap, flow;
    };
    vector<vector<Edge>> adj;
    vector<int> level, iter;
    
    MaxFlow(int n) : n(n), adj(n), level(n), iter(n) {}
    
    void addEdge(int from, int to, long long cap) {
        adj[from].push_back({to, (int)adj[to].size(), cap, 0});
        adj[to].push_back({from, (int)adj[from].size() - 1, 0, 0});
    }
    
    bool bfs(int s, int t) {
        fill(level.begin(), level.end(), -1);
        queue<int> q;
        level[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (auto& e : adj[u]) {
                if (e.cap - e.flow > 0 && level[e.to] < 0) {
                    level[e.to] = level[u] + 1;
                    q.push(e.to);
                }
            }
        }
        return level[t] >= 0;
    }
    
    long long dfs(int u, int t, long long f) {
        if (u == t) return f;
        for (int& i = iter[u]; i < adj[u].size(); i++) {
            Edge& e = adj[u][i];
            if (e.cap - e.flow > 0 && level[e.to] == level[u] + 1) {
                long long d = dfs(e.to, t, min(f, e.cap - e.flow));
                if (d > 0) {
                    e.flow += d;
                    adj[e.to][e.rev].flow -= d;
                    return d;
                }
            }
        }
        return 0;
    }
    
    long long maxFlow(int s, int t) {
        long long flow = 0;
        while (bfs(s, t)) {
            fill(iter.begin(), iter.end(), 0);
            long long d;
            while ((d = dfs(s, t, 1e18)) > 0) {
                flow += d;
            }
        }
        return flow;
    }
};
```

## 二分图

### 二分图判定（染色法）
```cpp
bool isBipartite(int n, vector<int> adj[]) {
    vector<int> color(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        if (color[i] == 0) {
            queue<int> q;
            color[i] = 1;
            q.push(i);
            while (!q.empty()) {
                int u = q.front();
                q.pop();
                for (int v : adj[u]) {
                    if (color[v] == 0) {
                        color[v] = 3 - color[u];
                        q.push(v);
                    } else if (color[v] == color[u]) {
                        return false;
                    }
                }
            }
        }
    }
    return true;
}
```

### 二分图最大匹配（匈牙利算法）
```cpp
// 时间复杂度：O(VE)
int hungarian(int n, int m, vector<int> adj[]) {
    vector<int> match(m + 1, 0);
    int result = 0;
    
    for (int u = 1; u <= n; u++) {
        vector<bool> visited(m + 1, false);
        if (dfs(u, adj, match, visited)) {
            result++;
        }
    }
    return result;
}

bool dfs(int u, vector<int> adj[], vector<int>& match, vector<bool>& visited) {
    for (int v : adj[u]) {
        if (!visited[v]) {
            visited[v] = true;
            if (match[v] == 0 || dfs(match[v], adj, match, visited)) {
                match[v] = u;
                return true;
            }
        }
    }
    return false;
}
```
