## unordered_map

### 头文件
```cpp
#include <unordered_map>
```

### 创建与初始化
```cpp
unordered_map<string, int> m;                         // 空表
unordered_map<string, int> m = {{"a", 1}, {"b", 2}}; // 列表初始化
unordered_map<string, int> m2(m1);                    // 拷贝构造
unordered_map<string, int> m(move(m1));               // 移动构造

// 自定义容量
unordered_map<string, int> m(100);                    // 预分配100个桶
unordered_map<string, int> m(100, hasher, keyEqual);  // 自定义哈希与比较
```

### 容量相关
```cpp
m.size();                  // 元素个数                    O(1)
m.empty();                 // 是否为空                    O(1)
m.max_size();              // 最大可容纳元素数            O(1)

m.bucket_count();          // 桶的数量                    O(1)
m.max_bucket_count();      // 最大桶数量                  O(1)
m.bucket_size(i);          // 第i个桶的元素个数           O(bucket_size)
m.load_factor();           // 负载因子 = size/bucket_count O(1)
m.max_load_factor();       // 最大负载因子（默认1.0）     O(1)
```

### 哈希策略
```cpp
m.rehash(n);               // 重新哈希，至少n个桶         O(n)
m.reserve(n);              // 预分配至少n个元素空间       O(n)
m.max_load_factor(0.75);   // 设置最大负载因子
```

### 访问与修改
```cpp
// 通过key访问（不存在则创建默认值）
m["key"] = value;          // 插入或修改                  均摊 O(1)
int val = m["key"];        // 不存在则创建0值             均摊 O(1)

// 通过at访问（不存在则抛异常）
m.at("key");               // 访问，不存在抛out_of_range  均摊 O(1)
m.at("key") = value;       // 修改                        均摊 O(1)

// insert
m.insert({"key", value});  // 插入键值对                  均摊 O(1)
m.insert(make_pair("key", value));
m.insert({{"a", 1}, {"b", 2}}); // 批量插入

// emplace（就地构造，更高效）
m.emplace("key", value);   // 就地构造                    均摊 O(1)
m.try_emplace("key", value); // C++17，仅不存在时插入     均摊 O(1)
m.insert_or_assign("key", value); // C++17，存在则赋值    均摊 O(1)

// erase
m.erase("key");            // 按key删除                   均摊 O(1)
m.erase(it);               // 按迭代器删除                均摊 O(1)
m.erase(first, last);      // 删除区间                    O(n)
m.clear();                 // 清空                        O(n)

// extract（C++17，不拷贝移动节点）
auto node = m.extract("key"); // 提取节点                 均摊 O(1)
if (!node.empty()) {
    node.key() = "new_key";
    m.insert(move(node));
}
m.merge(m2);               // 从m2移动元素到m             O(n)

// swap
m.swap(m2);                // 交换                        O(1)
```

### 查找操作
```cpp
// find
auto it = m.find("key");   // 查找，返回迭代器或end()     均摊 O(1)
if (it != m.end()) {
    cout << it->first << ": " << it->second;
}

// count
m.count("key");            // 返回0或1                    均摊 O(1)

// contains（C++20）
m.contains("key");         // 返回bool                    均摊 O(1)

// equal_range
auto [first, last] = m.equal_range("key"); // 返回范围    均摊 O(1)

// bucket操作
m.bucket("key");           // 返回key所在的桶编号
m.begin(i);                // 第i个桶的首迭代器
m.end(i);                  // 第i个桶的尾后迭代器
```

### 遍历
```cpp
// 范围for遍历
for (const auto& [key, value] : m) {
    cout << key << ": " << value << endl;
}

// 迭代器遍历
for (auto it = m.begin(); it != m.end(); ++it) {
    cout << it->first << ": " << it->second << endl;
}

// 遍历桶
for (size_t i = 0; i < m.bucket_count(); i++) {
    for (auto it = m.begin(i); it != m.end(i); ++it) {
        cout << it->first << ": " << it->second << endl;
    }
}
```

## unordered_set

### 头文件
```cpp
#include <unordered_set>
```

### 创建与初始化
```cpp
unordered_set<int> s;                           // 空集合
unordered_set<int> s = {1, 2, 3, 4, 5};        // 列表初始化
unordered_set<string> s({"apple", "banana"});
unordered_set<int> s2(s1);                      // 拷贝构造
```

### 常用操作
```cpp
// 容量
s.size();                // 元素个数
s.empty();               // 是否为空

// 插入
s.insert(val);           // 插入元素                    均摊 O(1)
s.insert({1, 2, 3});     // 批量插入
s.emplace(val);          // 就地构造                    均摊 O(1)

// 删除
s.erase(val);            // 删除元素                    均摊 O(1)
s.erase(it);             // 按迭代器删除
s.erase(first, last);    // 删除区间
s.clear();               // 清空

// 查找
s.find(val);             // 查找，返回迭代器            均摊 O(1)
s.count(val);            // 返回0或1                    均摊 O(1)
s.contains(val);         // C++20                       均摊 O(1)

// 遍历
for (const auto& x : s) {
    cout << x << " ";
}

// 哈希策略
s.bucket_count();        // 桶数量
s.load_factor();         // 负载因子
s.rehash(n);             // 重新哈希
s.reserve(n);            // 预分配空间
```

### 集合运算
```cpp
// 交集
unordered_set<int> intersection(const unordered_set<int>& a, const unordered_set<int>& b) {
    unordered_set<int> result;
    const auto& smaller = a.size() < b.size() ? a : b;
    const auto& larger = a.size() < b.size() ? b : a;
    for (int x : smaller) {
        if (larger.count(x)) result.insert(x);
    }
    return result;
}

// 并集
unordered_set<int> unionSet(const unordered_set<int>& a, const unordered_set<int>& b) {
    unordered_set<int> result(a);
    result.insert(b.begin(), b.end());
    return result;
}

// 差集
unordered_set<int> difference(const unordered_set<int>& a, const unordered_set<int>& b) {
    unordered_set<int> result;
    for (int x : a) {
        if (!b.count(x)) result.insert(x);
    }
    return result;
}
```

## unordered_multimap / unordered_multiset

```cpp
#include <unordered_map>
#include <unordered_set>

// 允许重复key
unordered_multimap<string, int> mm;
unordered_multiset<int> ms;

mm.insert({"key", 1});
mm.insert({"key", 2});      // 允许相同key

// 查找所有相同key
auto range = mm.equal_range("key");
for (auto it = range.first; it != range.second; ++it) {
    cout << it->second << " ";
}

mm.count("key");            // 返回key出现次数
mm.erase("key");            // 删除所有该key的元素
auto it = mm.find("key");   // 返回第一个匹配
```

## map

### 头文件
```cpp
#include <map>
```

### 创建与初始化
```cpp
map<string, int> m;                             // 空映射（默认按key升序）
map<string, int, greater<string>> m;            // 按key降序
map<string, int> m = {{"a", 1}, {"b", 2}};     // 列表初始化
map<string, int> m2(m1);                        // 拷贝构造
```

### 常用操作
```cpp
// 容量
m.size();                // 元素个数                    O(1)
m.empty();               // 是否为空                    O(1)

// 访问与修改
m["key"] = value;        // 插入或修改（不存在则创建）  O(log n)
m.at("key");             // 访问（不存在抛异常）        O(log n)

// insert
m.insert({"key", value}); // 插入                       O(log n)
m.emplace("key", value);  // 就地构造                   O(log n)
m.try_emplace("key", value); // C++17                  O(log n)
m.insert_or_assign("key", value); // C++17             O(log n)

// erase
m.erase("key");          // 按key删除                   O(log n)
m.erase(it);             // 按迭代器删除                均摊 O(1)
m.erase(first, last);    // 删除区间                    O(log n + k)

// 查找
m.find("key");           // 查找                        O(log n)
m.count("key");          // 0或1                        O(log n)
m.contains("key");       // C++20                       O(log n)
m.lower_bound("key");    // >= key的首个迭代器           O(log n)
m.upper_bound("key");    // > key的首个迭代器            O(log n)
m.equal_range("key");    // 返回范围对                  O(log n)

// 迭代器
m.begin();               // 首元素（最小key）
m.end();                 // 尾后
m.rbegin();              // 逆向首元素（最大key）
m.rend();                // 逆向尾后

// 交换
m.swap(m2);              // O(1)
```

### 遍历
```cpp
// 有序遍历（从小到大）
for (const auto& [key, value] : m) {
    cout << key << ": " << value << endl;
}

// 从大到小遍历
for (auto it = m.rbegin(); it != m.rend(); ++it) {
    cout << it->first << ": " << it->second << endl;
}
```

```cpp
#include <set>
```

### 常用操作
```cpp
set<int> s;                                 // 空集合（升序）
set<int, greater<int>> s;                   // 降序
set<int> s = {3, 1, 4, 1, 5};             // 自动去重 -> {1,3,4,5}

s.size();              // O(1)

s.erase(it);           // 按迭代器删除                  均摊 O(1)

s.count(val);          // 0或1                          O(log n)
s.contains(val);       // C++20                         O(log n)
s.lower_bound(val);    // >= val的首个元素               O(log n)
s.upper_bound(val);    // > val的首个元素                O(log n)
s.equal_range(val);    // 范围                          O(log n)

// 有序遍历
for (int x : s) cout << x << " ";
```

### 集合运算（STL算法）
```cpp
#include <algorithm>
#include <iterator>

set<int> a = {1, 2, 3, 4};
set<int> b = {3, 4, 5, 6};
set<int> result;

// 交集
set_intersection(a.begin(), a.end(), b.begin(), b.end(),
                 inserter(result, result.begin()));

          inserter(result, result.begin()));

               inserter(result, result.begin()));

```

## 自定义哈希函数

### 为自定义类型定义哈希
```cpp
struct Point {
    int x, y;
    bool operator==(const Point& other) const {
        return x == other.x && y == other.y;
    }
};

// 方式一：特化std::hash
namespace std {
template<>
struct hash<Point> {
    size_t operator()(const Point& p) const {
        return hash<int>()(p.x) ^ (hash<int>()(p.y) << 1);
    }
};
}

// 方式二：自定义哈希结构体
struct PointHash {
    size_t operator()(const Point& p) const {
        size_t h1 = hash<int>{}(p.x);
        size_t h2 = hash<int>{}(p.y);
        return h1 ^ (h2 << 1);
    }
};
struct PointEqual {
    bool operator()(const Point& a, const Point& b) const {
        return a.x == b.x && a.y == b.y;
    }
};
unordered_map<Point, string, PointHash, PointEqual> m;
```

### 常用类型的哈希组合
```cpp
// pair的哈希
struct PairHash {
    template<class T1, class T2>
    size_t operator()(const pair<T1, T2>& p) const {
        auto h1 = hash<T1>{}(p.first);
        auto h2 = hash<T2>{}(p.second);
        return h1 ^ (h2 << 1);
    }
};
unordered_map<pair<int, int>, int, PairHash> m;

// vector的哈希
struct VectorHash {
    size_t operator()(const vector<int>& v) const {
        size_t seed = v.size();
        for (auto& x : v) {
            seed ^= x + 0x9e3779b9 + (seed << 6) + (seed >> 2);
        }
        return seed;
    }
};
unordered_map<vector<int>, int, VectorHash> m;

// string的哈希（已有默认实现）
unordered_map<string, int> m; // 直接使用
```

### 高性能哈希（防哈希冲突攻击）
```cpp
#include <chrono>

struct CustomHash {
    static uint64_t splitmix64(uint64_t x) {
        x += 0x9e3779b97f4a7c15;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9;
        x = (x ^ (x >> 27)) * 0x94d049bb133111eb;
        return x ^ (x >> 31);
    }
    
    size_t operator()(uint64_t x) const {
        static const uint64_t FIXED_RANDOM = 
            chrono::steady_clock::now().time_since_epoch().count();
        return splitmix64(x + FIXED_RANDOM);
    }
};
unordered_map<long long, int, CustomHash> safeMap;
```

## 手动实现哈希表

### 链地址法
```cpp
template<typename K, typename V>
class HashTable {
private:
    struct Node {
        K key;
        V value;
        Node* next;
        Node(K k, V v) : key(k), value(v), next(nullptr) {}
    };
    
    vector<Node*> buckets;
    int size_;
    int capacity_;
    
    int hashFunc(const K& key) const {
        return hash<K>{}(key) % capacity_;
    }
    
    void rehash() {
        int oldCap = capacity_;
        vector<Node*> oldBuckets = buckets;
        capacity_ *= 2;
        buckets.assign(capacity_, nullptr);
        
        for (int i = 0; i < oldCap; i++) {
            Node* curr = oldBuckets[i];
            while (curr) {
                Node* next = curr->next;
                int newIdx = hashFunc(curr->key);
                curr->next = buckets[newIdx];
                buckets[newIdx] = curr;
                curr = next;
            }
        }
    }
    
public:
    HashTable(int cap = 16) : size_(0), capacity_(cap) {
        buckets.assign(capacity_, nullptr);
    }
    
    ~HashTable() {
        for (auto& head : buckets) {
            while (head) {
                Node* temp = head;
                head = head->next;
                delete temp;
            }
        }
    }
    
    void insert(const K& key, const V& value) {
        if (size_ >= capacity_) rehash();
        int idx = hashFunc(key);
        Node* curr = buckets[idx];
        while (curr) {
            if (curr->key == key) { curr->value = value; return; }
            curr = curr->next;
        }
        buckets[idx] = new Node(key, value);
        buckets[idx]->next = curr;
        size_++;
    }
    
    V* find(const K& key) {
        int idx = hashFunc(key);
        Node* curr = buckets[idx];
        while (curr) {
            if (curr->key == key) return &curr->value;
            curr = curr->next;
        }
        return nullptr;
    }
    
    bool erase(const K& key) {
        int idx = hashFunc(key);
        Node* curr = buckets[idx];
        Node* prev = nullptr;
        while (curr) {
            if (curr->key == key) {
                if (prev) prev->next = curr->next;
                else buckets[idx] = curr->next;
                delete curr;
                size_--;
                return true;
            }
            prev = curr;
            curr = curr->next;
        }
        return false;
    }
    
    int size() const { return size_; }
    bool empty() const { return size_ == 0; }
};
```

## 哈希与有序容器对比

| 特性 | unordered_map/set | map/set |
|------|-------------------|---------|
| 底层实现 | 哈希表 | 红黑树 |
| 插入 | 均摊 O(1) | O(log n) |
| 删除 | 均摊 O(1) | O(log n) |
| 查找 | 均摊 O(1) | O(log n) |
| 有序性 | 无序 | 有序 |
| 迭代器 | 前向迭代器 | 双向迭代器 |
| 最坏情况 | O(n) | O(log n) |
| 内存开销 | 较大（哈希表+桶） | 较大（树节点指针） |
| lower_bound | 无 | 有 |
| 适用场景 | 快速查找，不需有序 | 需要有序遍历、范围查询 |

### 选择建议
- **unordered_map/set**：只需快速插入/查找/删除，不需要有序
- **map/set**：需要有序遍历、范围查询（lower_bound/upper_bound）、有序统计
- **multimap/multiset**：允许重复且需要有序
- **unordered_multimap/multiset**：允许重复且需要快速查找
